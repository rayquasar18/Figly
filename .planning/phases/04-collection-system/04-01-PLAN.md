    - Category (id, name @unique, slug @unique, description?, coverImage?, position, createdAt, series[], followers[]) @@map("categories")
    - Series (id, categoryId, category, name, slug, description?, coverImage?, position, createdAt, items[], followers[]) @@unique([categoryId, slug]) @@index([categoryId]) @@map("series")

    - WishlistItem (id, userId, itemId, user, item, createdAt) @@unique([userId, itemId]) @@index([userId]) @@index([itemId]) @@map("wishlist_items")
    - Checklist (id, userId, user, name @db.VarChar(100), isPublic @default(false), createdAt, updatedAt, entries[]) @@index([userId]) @@map("checklists")
    - ChecklistEntry (id, checklistId, checklist, itemId?, item?, freeformText? @db.VarChar(200), isChecked @default(false), position @default(0), createdAt) @@index([checklistId, position]) @@map("checklist_entries")
    - PostItem (id, postId, itemId, post, item) @@unique([postId, itemId]) @@index([postId]) @@index([itemId]) @@map("post_items")
    - CategoryFollow (id, userId, categoryId, user, category, createdAt) @@unique([userId, categoryId]) @@index([userId]) @@map("category_follows")
    - SeriesFollow (id, userId, seriesId, user, series, createdAt) @@unique([userId, seriesId]) @@index([userId]) @@map("series_follows")
