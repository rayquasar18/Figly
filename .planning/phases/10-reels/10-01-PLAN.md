    private async processVideo(mediaId: string, originalKey: string): Promise<void> {
      const originalBuffer = await this.storageService.download(originalKey);
      const tmpDir = '/tmp';
      const tmpInput = path.join(tmpDir, `${mediaId}-input`);
      const tmpOutput = path.join(tmpDir, `${mediaId}-output.mp4`);
      const tmpThumb = path.join(tmpDir, `${mediaId}-thumb.jpg`);

      try {
        fs.writeFileSync(tmpInput, originalBuffer);

        // Probe video metadata (duration, dimensions)
        const metadata = await new Promise<any>((resolve, reject) => {
          ffmpeg.ffprobe(tmpInput, (err: any, data: any) => err ? reject(err) : resolve(data));
        });
        const videoStream = metadata.streams.find((s: any) => s.codec_type === 'video');
        const duration = metadata.format.duration || 0;
        const width = videoStream?.width || 0;
        const height = videoStream?.height || 0;

        // Transcode to H.264 baseline, AAC audio, max 720p short side
        await new Promise<void>((resolve, reject) => {
          ffmpeg(tmpInput)
            .outputOptions([
              '-c:v libx264',
              '-profile:v baseline',
              '-level 3.1',
              '-preset fast',
              '-crf 23',
              '-c:a aac',
              '-b:a 128k',
              '-movflags +faststart',
              '-vf', "scale='min(720,iw)':'min(1280,ih)':force_original_aspect_ratio=decrease",
            ])
            .output(tmpOutput)
            .on('end', () => resolve())
            .on('error', (err: any) => reject(err))
            .run();
        });

        // Generate thumbnail from first frame
        await new Promise<void>((resolve, reject) => {
          ffmpeg(tmpInput)
            .screenshots({
              count: 1,
              timemarks: ['0'],
              filename: `${mediaId}-thumb.jpg`,
              folder: tmpDir,
              size: '720x?',
            })
            .on('end', () => resolve())
            .on('error', (err: any) => reject(err));
        });

        // Upload transcoded video + thumbnail to MinIO
        const transcodedKey = originalKey.replace('originals/', 'transcoded/').replace(/\.[^.]+$/, '.mp4');
        const thumbnailKey = originalKey.replace('originals/', 'thumbnails/').replace(/\.[^.]+$/, '.jpg');

        await this.storageService.upload(transcodedKey, fs.readFileSync(tmpOutput), 'video/mp4');
        await this.storageService.upload(thumbnailKey, fs.readFileSync(tmpThumb), 'image/jpeg');

        // Update Media record with transcoded keys + reel metadata
        await this.prisma.media.update({
          where: { id: mediaId },
          data: {
            largeKey: transcodedKey,
            thumbnailKey: thumbnailKey,
            status: 'COMPLETED',
          },
        });

        // Store video metadata for ReelMeta creation (via a separate field or direct DB update)
        // Store duration/width/height in a JSON metadata approach or let PostsService read from media
        // For simplicity, store as a temporary record that PostsService reads
        // Actually: PostsService.createReel will ffprobe the original or we store metadata on Media
        // Best approach: store metadata directly on a new table or extend Media
        // Simplest: create a temporary JSON file or store in Redis
        // DECISION: Store in Media model's existing fields by adding metadata
        // We'll use a pragmatic approach: PostsService will read the video metadata from the thumbnail/transcoded keys
        // and the duration from ffprobe at reel creation time.
        // BUT: that requires downloading the file again. Instead, store metadata in a simple way:
        // Add the metadata to the media record using a JSON approach or just pass it through the job result.
        // FINAL APPROACH: Since we need duration/width/height for ReelMeta, store them via a simple
        // key-value in the database. We'll use the existing Media table and PostsService will
        // create the ReelMeta after the media is COMPLETED.
        // The cleanest approach: have the processor create a temporary metadata record.
        // SIMPLEST: Store metadata in /tmp as JSON, PostsService reads it when creating the reel.
        // ACTUALLY SIMPLEST: PostsService.createReel does its own ffprobe on the original video.
        // But original is in MinIO, not local. We'd need to download again.
        // PRAGMATIC: Store as a JSON metadata blob alongside the media update.
        // Let's use a metadata column or just hardcode a convention.
        //
        // REAL DECISION: Create ReelMeta at reel creation time. PostsService.createReel
        // receives duration/width/height from the frontend (which extracts from <video> element
        // loadedmetadata event) and stores it. This is the Instagram/TikTok pattern.
        // The processor only handles transcoding + thumbnail. Metadata comes from client.

        this.logger.log(`Video ${mediaId} transcoded successfully: ${duration}s, ${width}x${height}`);
      } finally {
        // Cleanup temp files
        [tmpInput, tmpOutput, tmpThumb].forEach(f => {
          try { if (fs.existsSync(f)) fs.unlinkSync(f); } catch {}
        });
      }
    }

    async createReel(userId: string, dto: CreateReelDto & { duration: number; width: number; height: number }) {
      // Validate mediaId belongs to user and is COMPLETED
      const media = await this.prisma.media.findFirst({
        where: { id: dto.mediaId, userId, status: 'COMPLETED' },
      });

      if (!media) {
        throw new BadRequestException('Video khong hop le hoac dang xu ly');
      }

      // Extract hashtags from caption
      const hashtags = this.extractHashtags(dto.caption || '');

      const post = await this.prisma.$transaction(async (tx: any) => {
        const post = await tx.post.create({
          data: {
            userId,
            caption: dto.caption,
            postType: 'REEL',
            media: {
              create: {
                mediaId: dto.mediaId,
                position: 0,
              },
            },
            reelMeta: {
              create: {
                duration: dto.duration,
                thumbnailKey: media.thumbnailKey,
                width: dto.width,
                height: dto.height,
              },
            },
          },
        });

        // Upsert hashtags and create links
        if (hashtags.length > 0) {
          for (const tag of hashtags) {
            const hashtag = await tx.hashtag.upsert({
              where: { name: tag },
              update: {},
              create: { name: tag },
            });
            await tx.postHashtag.create({
              data: { postId: post.id, hashtagId: hashtag.id },
            });
          }
        }

        // Create PostItem links for linked collection items
        if (dto.linkedItemIds && dto.linkedItemIds.length > 0) {
          for (const itemId of dto.linkedItemIds) {
            await tx.postItem.create({
              data: { postId: post.id, itemId },
            });
          }
        }

        return post;
      });

      return post;
    }
